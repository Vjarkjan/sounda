cmake_minimum_required(VERSION 3.20)
project(sounda_emulator_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT MELONDS_SOURCE)
  message(FATAL_ERROR "Pass -DMELONDS_SOURCE=/path/to/melonDS-1.1")
endif()
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT OFF CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB ON CACHE BOOL "" FORCE)
set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
add_subdirectory("${MELONDS_SOURCE}" "${CMAKE_BINARY_DIR}/melonDS")
find_package(PkgConfig REQUIRED)
pkg_check_modules(FAAD REQUIRED IMPORTED_TARGET faad2)
set(AAC_PLATFORM "${MELONDS_SOURCE}/src/frontend/qt_sdl/Platform_AAC.cpp")
set_source_files_properties("${AAC_PLATFORM}" PROPERTIES COMPILE_DEFINITIONS
  "AAC_Init=HostAAC_Init;AAC_DeInit=HostAAC_DeInit;AAC_Configure=HostAAC_Configure;AAC_DecodeFrame=HostAAC_DecodeFrame")
add_executable(sounda-emulator runner.cpp platform.cpp "${AAC_PLATFORM}")
target_compile_definitions(sounda-emulator PRIVATE GDBSTUB_ENABLED)
target_link_libraries(sounda-emulator PRIVATE core PkgConfig::FAAD ${CMAKE_DL_LIBS})
